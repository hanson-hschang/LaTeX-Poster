\newcommand{\raisedtri}[1]{%
  % 0.3ex is how high to raise it. 0.7 is the size (70%).
  % #1 passes the math style (display, text, etc.) so it shrinks in subscripts
  \raisebox{0.2ex}{\scalebox{0.6}{$#1\vartriangle$}}%
}
